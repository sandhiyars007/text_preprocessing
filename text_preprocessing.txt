stop_words = {
    "a","an","the","and","or","is","are","was","were","to","of","in",
    "on","for","with","that","this","them","using","can","be","which",
    "through","behind","different","it","its","their","as","by","from",
    "into","at","about","than","have","has","had","will","would","could",
    "should","do","does","did","not","but","if","then","else","while"
}
     

def clean_text(text):

    # Lowercase
    text = text.lower()

    # Remove emojis
    text = re.sub(r'[^\x00-\x7F]+', '', text)

    # Remove URLs
    text = re.sub(r'http\S+|www\S+', '', text)

    # Remove punctuation and numbers
    text = re.sub(r'[^a-zA-Z\s]', '', text)

    # Tokenization
    tokens = text.split()

    # Remove stopwords
    tokens = [word for word in tokens if word not in stop_words]

    # Join cleaned words
    cleaned_text = " ".join(tokens)

    return cleaned_text
     

df["cleaned_data"] = df["text"].apply(clean_text)

print(df[["text","cleaned_data"]])
